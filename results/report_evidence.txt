######## k8s/mongo-init.yaml ########
apiVersion: batch/v1
kind: Job
metadata:
  name: mongo-init
  namespace: emi
spec:
  backoffLimit: 4
  template:
    spec:
      restartPolicy: OnFailure
      containers:
        - name: mongo-init
          image: mongo:7
          command:
            - mongosh
            - --host
            - mongo-0.mongo
            - --quiet
            - --eval
            - |
              try {
                rs.status();
                print("Replica set already initiated");
              } catch (e) {
                printjson(rs.initiate({
                  _id: "rs0",
                  members: [
                    { _id: 0, host: "mongo-0.mongo:27017" },
                    { _id: 1, host: "mongo-1.mongo:27017" },
                    { _id: 2, host: "mongo-2.mongo:27017" }
                  ]
                }));
              }
              sleep(8000);
              rs.status().members.forEach(function (m) { print(m.name + " -> " + m.stateStr); });
